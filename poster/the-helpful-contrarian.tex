% ===== 01 · process =====
\psec{01}{Process}{Five Moves, In Order}

\begin{tcbraster}[raster columns=2, raster equal height=rows,
  raster column skip=7pt, raster row skip=7pt, raster valign=top]
\begin{stepcard}
\stepnum{01}\enspace\steptitle{Steel-Man First}\\[2pt]
\stepbody{Restate the proposal fairly \textbf{before} critiquing. List what is genuinely strong. Critiquing without understanding is straw-manning.}
\end{stepcard}
\begin{stepcard}
\stepnum{02}\enspace\steptitle{Audit Assumptions}\\[2pt]
\stepbody{List every unstated assumption. Rate each: \textbf{likelihood wrong} vs.\ \textbf{impact if wrong}. Focus on high-impact and uncertain; ignore the rest.}
\end{stepcard}
\begin{premcard}
\stepnum{03}\enspace\steptitle{Pre-Mortem}\enspace\badge{accent}{core}\\[2pt]
\stepbody{``It failed. Why?'' What assumption broke first? What early warning signs were missed? What cascaded?}
\end{premcard}
\begin{stepcard}
\stepnum{04}\enspace\steptitle{Invert Key Decisions}\\[2pt]
\stepbody{What if we did the \textbf{opposite}? What if we did \textbf{nothing}? Not every inversion is viable; the point is exposing hidden constraints.}
\end{stepcard}
\begin{stepcard}
\stepnum{05}\enspace\steptitle{Second-Order Effects}\\[2pt]
\stepbody{What happens \textit{after} what happens? Who else is affected that was not considered? What does this make harder in 6 months?}
\end{stepcard}
\begin{loopcard}
{\mono\fontsize{7.2}{9}\selectfont\color{white!70}\addfontfeatures{LetterSpace=8.0}THE CORE LOOP}\\[4pt]
{\mono\fontsize{8.2}{13}\selectfont\color{white}Understand \textcolor{accentsoft}{\arrow} find assumptions \textcolor{accentsoft}{\arrow} imagine it failed \textcolor{accentsoft}{\arrow} invert it \textcolor{accentsoft}{\arrow} trace the ripples}
\end{loopcard}
\end{tcbraster}

\psec{04}{Guardrails}{Anti-Patterns}
{\fontsize{8.8}{11.5}\selectfont\renewcommand{\arraystretch}{1.35}%
\rowcolors{2}{white}{paperalt}%
\begin{tabularx}{\linewidth}{@{}>{\bfseries\color{base700}}l >{\raggedright\arraybackslash}X@{}}
\rowcolor{base800}
\thd{Rule} & \multicolumn{1}{l@{}}{\thd{Practice}} \\
No empty objections & Reasoning \textit{and} an alternative \\
No vague doom & Name the specific failure \\
No straw-manning & Attack what was \textit{actually} proposed \\
No reverse bias & Always disagreeing is also biased \\
Label certainty & ``will break'' vs.\ ``might break if\ldots'' \\
Target the plan & Not the person \\
\end{tabularx}}
